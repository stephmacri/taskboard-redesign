// completion. The trip's action history (Offer to a carrier, Confirm Arrived, …)

// belongs to the trip and is read there, not here: mixing the two made the tab

// look like it was reporting on the trip rather than on this task.

function seedHistory(t){

  const who = ownerName(t.assignee) === 'Unassigned' ? 'Ops team' : ownerName(t.assignee);

  const at = t.updated || t.created;

  const h = [{a:'Task created', who:'System', when:t.created}];

  if(t.assignee) h.push({a:`Assigned to ${ownerName(t.assignee)}`, who:'System', when:t.created});

  if(t.status==='in_progress') h.push({a:'Task started', who, when:at});

  if(t.status==='waiting')     h.push({a:`Waiting on ${t.waitingOn || 'third party'}`, who, when:at});

  if(t.status==='done')        h.push({a:'Task completed', who, when:at});

  return h;

}

// Completed tasks are only ever revealed by the Status grouping's own toggle.

// Every other grouping shows outstanding work only — and because this is derived

// rather than stored, returning to Status restores whatever the user had chosen.

function completedVisible(){ return groupBy === 'status' && showPerformed; }


// The tab only means anything while the board is grouped by task type, so its

// scope is derived rather than stored — group some other way and the tab stops

// narrowing anything, come back and your tab is still where you left it.

function tabScope(){ return groupBy === 'type' ? activeTab : null; }


function taskMatchesFilters(t){

  const tab = tabScope();

  if(tab && t.type !== tab) return false;

  // A tab named "Missing approved POD" cannot hold a POD that is already

  // approved, so a type tab excludes completed work whatever the toggle says.

  // That is also what keeps each tab's count equal to what the board shows.

  if(tab && t.status==='done') return false;

  if(!completedVisible() && t.status==='done') return false;

  if(filters.status && t.status !== filters.status) return false;

  if(filters.assignee && (t.assignee || 'unassigned') !== filters.assignee) return false;

  if(filters.priority && t.priority !== filters.priority) return false;

  if(filters.due && dueBucket(t.hoursLeft) !== filters.due) return false;

  if(searchQuery){

    const owner = t.assignee ? (ASSIGNEE_NAMES[t.assignee] || t.assignee) : '';

    const haystack = [taskRef(t), t.order, t.title, t.desc, t.sub, owner, t.action].filter(Boolean).join(' ').toLowerCase();

    if(!haystack.includes(searchQuery)) return false;

  }

  return true;

}

// Render: board — columns come from whichever field is grouped on.

// With "Show performed" off, the Performed column disappears entirely

// under Status grouping, and performed tasks drop out of every other

// grouping too — not just hidden, genuinely out of the way.

// ---------------------------------------------------------------

function renderBoard(){

  const board = document.getElementById('board');

  board.innerHTML = '';

  const def = GROUP_FIELDS[groupBy];

  const filtered = TASKS.filter(taskMatchesFilters);


  // One column: the task-type group the tab bar is currently on.

  if(groupBy === 'type'){

    const label = activeTab ? TASK_TYPES[activeTab] : 'All tasks';

    board.appendChild(buildColumn({key:'__tab', label, color:'#8b8f84'}, filtered, null, true));

    return;

  }




  // Inside a type tab there is no completed work by definition, so that column

  // would sit permanently at zero — drop it rather than show a dead lane.

  const hideDone = groupBy==='status' && !showPerformed;

  const columns = hideDone ? def.columns.filter(c=>c.key!=='done') : def.columns;


  columns.forEach(colDef=>{

    const visible = filtered.filter(t=> def.valueOf(t) === colDef.key);

    board.appendChild(buildColumn(colDef, visible, def, false));

  });

}


// One column. `def` is the grouped-on field, or null when the column isn't a

// bucket of anything droppable (the single task-type column).

function buildColumn(colDef, visible, def, solo){

  const col = document.createElement('div');

  col.className = 'col' + (solo ? ' solo' : '');

  col.innerHTML = `

    <div class="col-head">

      <span class="dot" style="background:${colDef.color}"></span>

      <h3>${colDef.label}</h3>

      <span class="count">${visible.length}</span>

    </div>

    <div class="col-body"></div>

  `;

  const body = col.querySelector('.col-body');

  if(visible.length===0){

    body.innerHTML = `<div style="font-size:11.5px;color:var(--text-3);padding:10px 4px;">Nothing matches.</div>`;

  }

  // Cards are ordered by deadline, not by priority: whatever is already

  // overdue or closest to its due date sits at the top, and tasks with no

  // deadline sink to the bottom. Priority only breaks ties between two tasks

  // that share the same deadline — it never outranks the clock.

  visible

    .sort((a,b)=>{

      const da = a.hoursLeft===null ? Infinity : a.hoursLeft;

      const db = b.hoursLeft===null ? Infinity : b.hoursLeft;

      if(da !== db) return da - db;

      return PRIORITY_ORDER.indexOf(a.priority) - PRIORITY_ORDER.indexOf(b.priority);

    })

    .forEach(t=> body.appendChild(renderCard(t)));


  if(def && def.setValue){

    body.addEventListener('dragover', (e)=>{ e.preventDefault(); body.classList.add('drag-over'); });

    body.addEventListener('dragleave', ()=> body.classList.remove('drag-over'));

    body.addEventListener('drop', (e)=>{

      e.preventDefault();

      body.classList.remove('drag-over');

      const task = TASKS.find(x=>x.id===e.dataTransfer.getData('text/plain'));

      if(!task) return;

      if(def.valueOf(task) === colDef.key) return; // dropped back where it started

      def.setValue(task, colDef.key);

      renderBoard();

      toast(`Moved ${taskRef(task)} to "${colDef.label}"`);

    });

  }

  return col;

}


// Card anatomy, top to bottom: title, trip reference, task description,

// "Mark as completed" closes the task out from here without a trip round-trip —

// for the cases users described where the work already happened somewhere else

// (a call, an email) and only the task is still open. The panel stays on the

// task so the result of completing it is visible rather than just vanishing.

document.getElementById('completeTaskBtn').addEventListener('click', ()=>{

  const t = panelTask;

  if(!t || t.status==='done') return;

  t.status = 'done';

  t.hoursLeft = null;

  t.history.push({a:t.action, who:ownerName(t.assignee || CURRENT_USER.code), when:'Just now'});

  renderBoard();

  openPanel(t);

  toast(`${taskRef(t)} marked as completed`);

});


document.getElementById('panelClose').addEventListener('click', closePanel);

function closePanel(){

  panelTask = null;

  document.getElementById('panelContent').classList.remove('show');

  document.getElementById('panelEmpty').style.display = 'flex';

}


// ---------------------------------------------------------------

// Toast

// ---------------------------------------------------------------

let toastTimer;

function toast(msg){

  const el = document.getElementById('toast');

  el.textContent = msg;

  el.classList.add('show');

  clearTimeout(toastTimer);

  toastTimer = setTimeout(()=> el.classList.remove('show'), 2600);

}
